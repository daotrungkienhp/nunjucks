{% macro renderVideoPlayer(video) %}
  <div class="bem-video__player" x-data="{ playing: false }">
    {% if video.type == "youtube" %}
      <div class="bem-video__embed">
        {% if video.thumbnail %}
          <img src="{{ video.thumbnail }}" alt="{{ video.title or 'Video' }}" class="bem-video__thumbnail" loading="lazy" />
          <button @click="playing = true" class="bem-video__play-btn">
            <svg viewBox="0 0 24 24" fill="currentColor" class="w-12 h-12 md:w-16 md:h-16">
              <path d="M8 5v14l11-7z"/>
            </svg>
          </button>
        {% endif %}
        <template x-if="playing">
          <iframe src="{{ video.url | replace('watch?v=', 'embed/') }}?autoplay=1"
                  class="bem-video__iframe" allow="autoplay; encrypted-media" allowfullscreen loading="lazy"></iframe>
        </template>
      </div>
    {% elif video.type == "vimeo" %}
      <div class="bem-video__embed">
        {% if video.thumbnail %}
          <img src="{{ video.thumbnail }}" alt="{{ video.title or 'Video' }}" class="bem-video__thumbnail" loading="lazy" />
          <button @click="playing = true" class="bem-video__play-btn">
            <svg viewBox="0 0 24 24" fill="currentColor" class="w-12 h-12 md:w-16 md:h-16">
              <path d="M8 5v14l11-7z"/>
            </svg>
          </button>
        {% endif %}
        <template x-if="playing">
          <iframe src="https://player.vimeo.com/video/{{ video.id }}?autoplay=1"
                  class="bem-video__iframe" allow="autoplay" allowfullscreen loading="lazy"></iframe>
        </template>
      </div>
    {% else %}
      <video class="bem-video__file" controls preload="metadata" {% if video.poster %}poster="{{ video.poster }}"{% endif %}>
        <source src="{{ video.url }}" type="{{ video.mime or 'video/mp4' }}">
      </video>
    {% endif %}
  </div>
{% endmacro %}

{% macro renderSlider(items=[]) %}
  {% set slides = items %}
  {% set totalSlides = slides | length %}
  
  {% if totalSlides > 0 %}
    <section class="bem-slider group" 
        x-data='{ 
            currentIndex: 3,
            visibleCount: 1,
            autoPlayInterval: null,
            originalCount: {{ totalSlides }},
            isTransitioning: false,
            
            init() {
                this.updateVisibleCount();
                window.addEventListener("resize", () => this.updateVisibleCount());
                this.startAutoPlay();
            },
            updateVisibleCount() {
                if (window.innerWidth >= 1024) {
                    this.visibleCount = 3;
                } else if (window.innerWidth >= 768) {
                    this.visibleCount = 2;
                } else {
                    this.visibleCount = 1;
                }
            },
            next() {
                if (this.isTransitioning) return;
                this.currentIndex++;
                this.checkJump();
            },
            prev() {
                if (this.isTransitioning) return;
                this.currentIndex--;
                this.checkJump();
            },
            checkJump() {
                setTimeout(() => {
                    if (this.currentIndex >= 3 + this.originalCount) {
                        this.isTransitioning = true;
                        this.currentIndex = 3;
                        setTimeout(() => { this.isTransitioning = false }, 50);
                    } else if (this.currentIndex < 3) {
                        this.isTransitioning = true;
                        this.currentIndex = 3 + this.originalCount - 1;
                        setTimeout(() => { this.isTransitioning = false }, 50);
                    }
                }, 700);
            },
            startAutoPlay() {
                this.autoPlayInterval = setInterval(() => {
                    this.next();
                }, 5000);
            },
            stopAutoPlay() {
                clearInterval(this.autoPlayInterval);
            }
        }'
        @mouseenter="stopAutoPlay()"
        @mouseleave="startAutoPlay()">
        
        <div class="bem-slider__wrapper">
            <!-- Slider Container -->
            <div class="bem-slider__track"
                 :class="isTransitioning ? 'transition-none' : 'transition-transform duration-700 ease-in-out'"
                 :style="'transform: translateX(-' + (currentIndex * (100 / visibleCount)) + '%)'">
                
                {# Render clones and original slides. Loop from 0 to totalSlides + 5 #}
                {% for i in range(0, totalSlides + 6) %}
                    {% set slideIndex = (i - 3 + totalSlides * 2) % totalSlides %}
                    {% set slide = slides[slideIndex] %}
                    {{ renderSlideItem(slide, i, visibleCount) }}
                {% endfor %}
            </div>

            {{ renderSliderNav() }}
            {{ renderSliderDots() }}
        </div>
    </section>
  {% endif %}
{% endmacro %}

{% macro renderStatItem(stat) %}
  <div class="bem-stats__item" x-data="{ count: 0, target: {{ stat.value }}, unit: '{{ stat.unit or "" }}' }"
       x-intersect="
         let start = 0;
         let duration = 2000;
         let step = Math.ceil(target / (duration / 16));
         let timer = setInterval(() => {
           start += step;
           if (start >= target) { start = target; clearInterval(timer); }
           count = start;
         }, 16);
       ">
    <span class="bem-stats__value"><span x-text="count.toLocaleString()">0</span>{{ stat.unit or "" }}</span>
    {% if stat.label %}
      <span class="bem-stats__label">{{ stat.label }}</span>
    {% endif %}
  </div>
{% endmacro %}
